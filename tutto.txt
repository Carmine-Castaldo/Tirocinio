class DecisionLogger {
public:
    static DecisionLogger& get() { static DecisionLogger l; return l; }

    void open(const std::string& path) {
        out_.open(path, std::ios::out | std::ios::trunc);
        if (out_.is_open()) out_ << "t,winner,e_winner,competitors\n";
    }

    void log(double t, const std::string& winner, double eWinner,
             const std::vector<std::pair<std::string, double>>& competitors) {
        if (!out_.is_open()) return;
        out_ << std::fixed << std::setprecision(2) << t << ',' << winner << ',' << eWinner << ",\"";
        for (size_t i = 0; i < competitors.size(); ++i) {
            if (i) out_ << ';';
            out_ << competitors[i].first << '=' << competitors[i].second;
        }
        out_ << "\"\n";
    }

    void close() { if (out_.is_open()) out_.close(); }

private:
    std::ofstream out_;
};

#pragma once
#include <unordered_map>
#include <unordered_set>
#include <vector>
#include <tuple>
#include "RemoteAPIObjects.h"

class SharedPerception {
public:
    using Sonar = std::tuple<int64_t, double, std::vector<double>, int64_t, std::vector<double>>;

    static SharedPerception& get() { static SharedPerception s; return s; }

    void setup(RemoteAPIObjects* sim, int64_t robot, int64_t camera) { sim_ = sim; robot_ = robot; camera_ = camera; }
    void newTick() { poseOk_ = false; treeOk_ = false; sonar_.clear(); goal_ = bestGoal_;                              // il goal del tick precedente (ritardo di 1 tick: irrilevante)
                     bestGoal_ = -1; bestE_ = -1.0; timeOk_ = false;}

    const std::vector<double>& position()    { refreshPose(); return pos_; }
    const std::vector<double>& orientation() { refreshPose(); return ori_; }
    const int64_t cameraHandle(){ return camera_;}

    const Sonar& sonar(int64_t h) { 
        auto it = sonar_.find(h);
        if (it == sonar_.end()) it = sonar_.emplace(h, sim_->sim().readProximitySensor(h)).first;
        return it->second;
    }

    bool inRobotTree(int64_t h) { 
        if (!treeOk_) {
            tree_.clear();
            for (auto x : sim_->sim().getObjectsInTree(robot_)) tree_.insert(x);
            treeOk_ = true;
        }
        return tree_.count(h) > 0;
    }

    void offerGoal(int64_t h, double e) {
        if (h >= 0 && e > bestE_) { bestE_ = e; bestGoal_ = h; }
    }
    bool isGoal(int64_t det) {
        if (goal_ < 0 || det < 0) return false;
        if (det == goal_) return true;
        if (goalTreeFor_ != goal_) {
            goalTree_.clear();
            try { for (auto x : sim_->sim().getObjectsInTree(goal_)) goalTree_.insert(x); } catch (...) {}
            goalTreeFor_ = goal_;
        }
        return goalTree_.count(det) > 0;
    }
    double time() {
        if (!timeOk_) { t_ = sim_->sim().getSimulationTime(); timeOk_ = true; }
        return t_;
    }

private:
    void refreshPose() {
        if (poseOk_) return;
        pos_ = sim_->sim().getObjectPosition(robot_, -1);
        ori_ = sim_->sim().getObjectOrientation(robot_, -1);
        poseOk_ = true;
    }
    RemoteAPIObjects* sim_ = nullptr;
    int64_t robot_ = -1, camera_ = -1;
    bool poseOk_ = false, treeOk_ = false;
    std::vector<double> pos_, ori_;
    std::unordered_map<int64_t, Sonar> sonar_;
    std::unordered_set<int64_t> tree_;
    int64_t goal_ = -1, bestGoal_ = -1, goalTreeFor_ = -2;
    double  bestE_ = -1.0;
    double t_ = 0.0; bool timeOk_ = false;
    std::unordered_set<int64_t> goalTree_;
};
===== ./headers/TaskPriorityManager.h =====
#pragma once
#include <behaviortree_cpp/blackboard.h>
#include <vector>
#include <string>
#include <iostream>
#include <iomanip>

struct TaskStep {
    std::string name;     
    std::string mu_port;  
    std::string done_port;
};

class TaskPriorityManager {
public:
    TaskPriorityManager(std::string taskName,
                        BT::Blackboard::Ptr blackboard,
                        std::vector<TaskStep> sequence,
                        double baseMu = 1.0,
                        double prioritizedBonus = 1.0) 
        : taskName_(std::move(taskName)),
          blackboard_(std::move(blackboard)),
          sequence_(std::move(sequence)),
          baseMu_(baseMu),
          prioritizedBonus_(prioritizedBonus)
    {
        update();
    }

    void update() {
        int firstUnfinishedIdx = -1;

        for (size_t i = 0; i < sequence_.size(); ++i) {
            bool isDone = false;
            if (blackboard_->get(sequence_[i].done_port, isDone) && isDone) {
                continue;
            } else {
                firstUnfinishedIdx = static_cast<int>(i);
                break;
            }
        }


        if (firstUnfinishedIdx != currentActiveIndex_) {
            currentActiveIndex_ = firstUnfinishedIdx;

            if (firstUnfinishedIdx != -1) {
                double currentMu = baseMu_;
                (void)blackboard_->get(sequence_[firstUnfinishedIdx].mu_port, currentMu);

                double newMu = currentMu + prioritizedBonus_;
                std::cout << currentMu << std::endl;
                blackboard_->set(sequence_[firstUnfinishedIdx].mu_port, newMu);

                std::cout << "\n================================================================" << std::endl;
                if (firstUnfinishedIdx == 0) {
                    std::cout << "[" << taskName_ << " - START] Priorizited Mangitude To First Behavior In Sequence: "
                              << sequence_[firstUnfinishedIdx].name << " (mu: " << currentMu << " -> " << newMu << ")" << std::endl;
                } else {
                    std::cout << "[" << taskName_ << " - STEP TERMINATO] Incremented Magnitude To Next Behavior In Sequence: "
                              << sequence_[firstUnfinishedIdx].name << " (magnitude: " << currentMu << " -> " << newMu << ")" << std::endl;
                }
                std::cout << "----------------------------------------------------------------" << std::endl;
                printCurrentMuState();
                std::cout << "================================================================\n" << std::endl;
            } else {
                std::cout << "\n[" << taskName_ << " - COMPLETED] !\n" << std::endl;
            }
        }
    }

private:
    void printCurrentMuState() const {
        for (size_t i = 0; i < sequence_.size(); ++i) {
            bool isDone = false;
            (void)blackboard_->get(sequence_[i].done_port, isDone);

            double muVal = baseMu_;
            (void)blackboard_->get(sequence_[i].mu_port, muVal);

            std::string status = isDone ? "COMPLETED" : (static_cast<int>(i) == currentActiveIndex_ ? ">>> ACTIVE BEHAVIOR <<<" : "TO COMPLETE");
            std::cout << "  Step [" << (i + 1) << "] " << std::left << std::setw(15) << sequence_[i].name
                      << " | Port: " << std::setw(8) << sequence_[i].mu_port
                      << " | magnitude = " << muVal << "  (" << status << ")" << std::endl;
        }
    }

    std::string taskName_;
    BT::Blackboard::Ptr blackboard_;
    std::vector<TaskStep> sequence_;
    double baseMu_;
    double prioritizedBonus_;
    int currentActiveIndex_ = -2;
};

#pragma once
#include <cmath>
#include <random>
#include <stdexcept>
#include <string>
#include <vector>
#include "RemoteAPIObjects.h"

inline unsigned& trialSeed() { static unsigned s = 1; return s; }

struct CubePose { std::string path; double x, y; };

inline std::vector<CubePose> randomizeCubes(RemoteAPIObjects& sim, unsigned seed,
                                            const std::vector<std::string>& cubePaths,
                                            const std::vector<std::string>& keepOutPaths,
                                            double half = 6.5,
                                            double minCubeDist = 1.5,
                                            double minKeepOut = 2.0)
{
    std::mt19937 rng(seed);
    std::uniform_real_distribution<double> U(-half, half);

    std::vector<std::pair<double, double>> keep;
    for (const auto& p : keepOutPaths) {
        auto pos = sim.sim().getObjectPosition(sim.sim().getObjectHandle(p), -1);
        keep.emplace_back(pos[0], pos[1]);
    }

    std::vector<CubePose> placed;
    for (const auto& path : cubePaths) {
        int64_t h = sim.sim().getObjectHandle(path);
        auto cur = sim.sim().getObjectPosition(h, -1);
        bool ok = false;
        for (int attempt = 0; attempt < 2000 && !ok; ++attempt) {
            const double x = U(rng), y = U(rng);
            ok = true;
            for (const auto& k : keep)   if (std::hypot(x - k.first, y - k.second) < minKeepOut) ok = false;
            for (const auto& c : placed) if (std::hypot(x - c.x, y - c.y) < minCubeDist)         ok = false;
            if (ok) {
                placed.push_back({path, x, y});
                sim.sim().setObjectPosition(h, -1, {x, y, cur[2]});
            }
        }
        if (!ok) throw std::runtime_error("randomizeCubes: nessuna posizione valida per " + path);
    }
    return placed;
}

    const std::string out = (argc > 2) ? argv[2] : "results";

SimHandles initSimulation(RemoteAPIObjects& sim) {

BT::Tree createBehaviorTree(BT::BehaviorTreeFactory& factory, RemoteAPIObjects& sim, const std::string& xmlPath) {
    factory.registerNodeType<AvoidObstacles>("AvoidObstacles", &sim);
    factory.registerNodeType<ReachColor>("ReachColor", &sim);

void setupEnvironmentAndBlackboard(RemoteAPIObjects& sim, const BT::Blackboard::Ptr& blackboard, unsigned seed, const std::string& outDir) {
    auto cubes = randomizeCubes(sim, seed,
                                objectsPath_,
                                {robotPath_, humanPath_});
    {
        std::ofstream f(outDir + "/cubes_seed" + std::to_string(seed) + ".csv");
        f << "cube,x,y\n";
        for (const auto& c : cubes) {
            f << c.path << ',' << c.x << ',' << c.y << '\n';
        }
    }

    for (const auto& c : cubes) {   
        std::string name = c.path.substr(1);
        blackboard->set(name + "_x", c.x);
        blackboard->set(name + "_y", c.y);
    }

    int64_t billHandle = sim.sim().getObjectHandle(humanPath_);
    auto posBill = sim.sim().getObjectPosition(billHandle, -1);
    blackboard->set("Bill_x", posBill[0]);
    blackboard->set("Bill_y", posBill[1]);
}

std::pair<std::string, double> runSimulationLoop(RemoteAPIClient& client, RemoteAPIObjects& sim, BT::Tree& tree, 
                                                 const SimHandles& handles, double maxT) {
    std::string result = "TIMEOUT";
    double pathLen = 0.0, px = 0.0, py = 0.0;
    bool havePrev = false;

    while (true) {
        SharedPerception::get().newTick();
        sim.sim().handleVisionSensor(handles.camera);
        
        const double t = SharedPerception::get().time();
        const auto& pos = SharedPerception::get().position();
        
        if (havePrev) pathLen += std::hypot(pos[0] - px, pos[1] - py);
        px = pos[0]; 
        py = pos[1]; 
        havePrev = true;
        
        MotorArbitrator::getInstance().reset();
        BT::NodeStatus st;
        try { st = tree.tickExactlyOnce(); }
        catch (const std::exception& e) { std::cout << "[BT ERROR] " << e.what() << std::endl; result = "ERROR"; break; }
        MotorArbitrator::getInstance().applyCommand(&sim, handles.leftMotor, handles.rightMotor);
        client.step();

        if (st != BT::NodeStatus::RUNNING) {
            result = (st == BT::NodeStatus::SUCCESS) ? "SUCCESS" : "FAILURE";
            break;
        }
        if (t > maxT) break;
    }

    return {result, pathLen};
}

void logFailures(const BT::Tree& tree, const BT::TreeObserver& observer, const std::string& outDir, unsigned seed) {
    std::vector<std::tuple<long long, std::string, unsigned>> rows;
    tree.applyVisitor([&](const BT::TreeNode* n) {
        const auto& ns = observer.getStatistics(n->UID());
        if (ns.last_result == BT::NodeStatus::FAILURE) {
            rows.emplace_back(ns.last_timestamp.count(), n->fullPath(), ns.failure_count);
        }
    });
    
    std::sort(rows.rbegin(), rows.rend());
    std::ofstream f(outDir + "/failures_seed" + std::to_string(seed) + ".txt");
    for (size_t i = 0; i < rows.size() && i < 15; ++i) {
        f << std::get<0>(rows[i]) << " ms  " << std::get<1>(rows[i])
          << "  failures=" << std::get<2>(rows[i]) << '\n';
    }
}

void saveTrialSummary(const std::string& outDir, unsigned seed, const std::string& result, double simTime, double pathLen) {
    const std::string summary = outDir + "/summary.csv";
    const bool exists = std::ifstream(summary).good();
    std::ofstream s(summary, std::ios::app);

    if (!exists) s << "seed,result,sim_time_s,path_m\n";
    s << seed << ',' << result << ',' << simTime << ',' << pathLen << '\n';
    
    std::cout << "[TRIAL " << seed << "] " << result 
              << " t=" << simTime << "s path=" << pathLen << "m\n";
}

BT::NodeStatus AttentionalBehavior::tick() {
    const double now = SharedPerception::get().time();

    getInput("emphasis", emphasis_); 
    getInput("period",   period_);
    if (emphasis_ <= 0.0) emphasis_ = 1e-3;
    if (period_   <= 0.0) period_   = 1.0;

    BT::NodeStatus currentStatus = BT::NodeStatus::RUNNING;

    if (now >= lastActivation_ + period_) { 
        lastActivation_ = now;
        setOutput("stimuli", readSensors());
        double ext = 1.0; getInput("external", ext);
        active_ = checkInternalReleaser() && (ext > 0.5);
        if (active_) {
            std::tie(lastVL_, lastVR_) = executeMotorSchema();
            const bool stopped = (lastVL_ == 0.0 && lastVR_ == 0.0);
            currentStatus = (stopped && name() != "avoid") ? BT::NodeStatus::SUCCESS
                                                           : BT::NodeStatus::RUNNING;
        } else { lastVL_ = lastVR_ = 0.0; }
    }
    if (active_) {
        MotorArbitrator::getInstance().proposeCommand(getUniqueID(), emphasis_, lastVL_, lastVR_, exclusive());
        afterPropose();
    }
    return currentStatus;
}

void AttentionalBehavior::halt() {
    lastActivation_ = -1e9; 
    lastVL_ = 0.0;
    lastVR_ = 0.0;
    active_ = false;
    resetStatus();
}

#include "AvoidObstacles.h"
#include "SharedPerception.h"
#include <algorithm>
#include <cmath>

AvoidObstacles::AvoidObstacles(const std::string& name, const BT::NodeConfig& config, RemoteAPIObjects* sim)
    : AttentionalBehavior(name, config), sim_(sim) {
    robotHandle_ = sim_->sim().getObjectHandle("/PioneerP3DX");

    for (int i = 1; i <= 6; ++i) {
        int64_t h = sim_->sim().getObjectHandle(
            "/PioneerP3DX/ultrasonicSensor[" + std::to_string(i) + "]");
        sonarHandles_.push_back(h);

        auto p = sim_->sim().getObjectPosition(h, robotHandle_);
        double a = std::atan2(p[1], p[0]);
        sonarSide_.push_back((a > 0.35) ? +1 : (a < -0.35 ? -1 : 0));
    }
    lastDistances_.assign(sonarHandles_.size(), -1.0);
}

BT::PortsList AvoidObstacles::providedPorts() {
    BT::PortsList ports = {
        BT::InputPort<std::string>("Object")
    };
    auto basePorts = AttentionalBehavior::providedPorts();
    ports.insert(basePorts.begin(), basePorts.end());
    return ports;
}

bool AvoidObstacles::isIgnored(int64_t detected) {
    if (detected == -1) return false;
    if (detected == targetHandle_) return true;
    if (!ignoredValid_) {
        ignored_.clear();
        try {
            for (auto h : sim_->sim().getObjectsInTree(robotHandle_)) ignored_.insert(h);
        } catch (...) {}
        ignoredValid_ = true;
    }
    return ignored_.count(detected) > 0;
}

double AvoidObstacles::checkSonar(int64_t h) {
    const auto& r = SharedPerception::get().sonar(h);
    if (std::get<0>(r) == 1) {
        int64_t det = std::get<3>(r);
        if (!SharedPerception::get().isGoal(det) &&
            !SharedPerception::get().inRobotTree(det))
            return std::get<1>(r);
    }
    return -1;
}

double AvoidObstacles::readSensors() {
    std::string path;
    getInput("Object", path);
    if (path != previousPath_) {
        targetHandle_ = -1;
        if (path.empty())
            previousPath_ = path;
        else {
            try { targetHandle_ = sim_->sim().getObjectHandle(path); previousPath_ = path; }
            catch (...) {}
        }
    }
    ignoredValid_ = false;
    distances.assign(sonarHandles_.size(), -1.0);
    for (size_t i = 0; i < sonarHandles_.size(); ++i)
        distances[i] = checkSonar(sonarHandles_[i]);

    isObstacle = false;
    double minDistance = 999.0;
    for (double d : distances) {
        if (d > 0.0) {
            isObstacle = true;
            if (d < minDistance) minDistance = d;
        }
    }

    const double t = SharedPerception::get().time();
    if (isObstacle) { 
        holdUntil_ = t + kHoldSec; 
        lastDistances_ = distances; 
        lastMinDist_ = minDistance; 
    } else if (t >= holdUntil_) {
        turnSign_ = 0;
        lastMinDist_ = 999.0; 
    }

    return isObstacle ? minDistance : lastMinDist_;
}

bool AvoidObstacles::checkInternalReleaser() {
    return isObstacle || SharedPerception::get().time() < holdUntil_;
}

std::pair<double,double> AvoidObstacles::executeMotorSchema() {
    const auto& d = isObstacle ? distances : lastDistances_;

    double turn = 0.0, minDist = 999.0, minCenter = 999.0, leftSum = 0.0, rightSum = 0.0;
    for (size_t i = 0; i < d.size(); ++i) {
        if (d[i] <= 0.0) continue;
        minDist = std::min(minDist, d[i]);
        double w = 1.0 / (d[i] + 0.20);
        if (sonarSide_[i] > 0)      { turn -= w; leftSum  += w; }
        else if (sonarSide_[i] < 0) { turn += w; rightSum += w; }
        else                        { minCenter = std::min(minCenter, d[i]); }
    }

    bool symmetric = std::abs(turn) < 0.3 * std::max(leftSum, rightSum);
    if (minCenter < 999.0 || symmetric) {
        if (turnSign_ == 0) turnSign_ = (leftSum > rightSum) ? -1 : +1;
        turn += turnSign_ * (1.0 / (std::min(minCenter, minDist) + 0.20));
    }

    double v = 0.35;
    if (minDist < 0.20)      v = 0.05;
    else if (minDist < 0.40) v = 0.18;

    double w_gain = (minDist < 0.35) ? 0.55 : 0.35;
    double w = std::clamp(turn * w_gain, -1.2, 1.2);

    const double L = 0.331;
    return { v - w * L / 2.0, v + w * L / 2.0 };
}

double GoToXY::readSensors() {
    getInput("X", targetX_);
    getInput("Y", targetY_);

    const auto& robotPos = SharedPerception::get().position();
    const auto& robotOri = SharedPerception::get().orientation();

    double dx = targetX_ - robotPos[0];
    double dy = targetY_ - robotPos[1];
    
    dist_ = std::hypot(dx, dy);

    double targetAngle = std::atan2(dy, dx);
    angleError_ = targetAngle - robotOri[2];
    
    while (angleError_ > M_PI)  angleError_ -= 2.0 * M_PI;
    while (angleError_ < -M_PI) angleError_ += 2.0 * M_PI;

    return dist_;
}

bool GoToXY::checkInternalReleaser() { 
    if (std::abs(targetX_) > 7.5 || std::abs(targetY_) > 7.5) return false;     
    return true; 
}

std::pair<double, double> GoToXY::executeMotorSchema() { 
    if (dist_ <= 0.55) return {0.0, 0.0};
    double v = 0.25 * std::max(0.0, std::cos(angleError_));
    double w = std::clamp(angleError_ * 1.0, -0.8, 0.8);

void MotorArbitrator::reset() {
    proposals_.clear();
    currentWinner_ = "none";
    maxEmphasis_ = -1.0;
    winningVLeft_ = 0.0;
    winningVRight_ = 0.0;
}

void MotorArbitrator::proposeCommand(const std::string& id, double e,
                                     double vL, double vR, bool exclusive) {
    if (e <= 0.0) return;
    proposals_.push_back({id, e, vL, vR, exclusive});
}

bool MotorArbitrator::getWinningCommand(double& vLeft, double& vRight, std::string& winnerName, double& winnerEmphasis) const {
    if (maxEmphasis_ < 0.0) 
        return false;
    
    vLeft = winningVLeft_;
    vRight = winningVRight_;
    winnerName = currentWinner_;
    winnerEmphasis = maxEmphasis_;
    // std::cout << "[MotorArbitrator] Comando finale applicato -> Vincitore: " << currentWinner_ 
    //           << " (Enfasi: " << maxEmphasis_ << ") | Velocita': (" << winningVLeft_ << ", " << winningVRight_ << ")" << std::endl;

    return true;
}

void MotorArbitrator::applyCommand(RemoteAPIObjects* sim, int64_t leftMotor, int64_t rightMotor) {
    const Proposal* win = nullptr;
    
    for (auto& p : proposals_)
        if (p.exclusive && (!win || p.emphasis > win->emphasis)) win = &p;
    std::unordered_map<std::string, const Proposal*> uniq;
    for (auto& p : proposals_)
        if (!p.exclusive) { auto& s = uniq[p.id]; if (!s || p.emphasis > s->emphasis) s = &p; }

    double sE = 0, sL = 0, sR = 0; const Proposal* top = nullptr;

    auto add = [&](const Proposal* p) {
        sE += p->emphasis; sL += p->emphasis * p->vL; sR += p->emphasis * p->vR;
        if (!top || p->emphasis > top->emphasis) top = p;
    };
    if (win) add(win);
    for (auto& kv : uniq) add(kv.second);

    double vL = sE > 0 ? sL / sE : 0.0, vR = sE > 0 ? sR / sE : 0.0;
    const std::string winner = top ? top->id : "none";

    if (winner != previousWinner_) {
            previousWinner_ = winner;
            std::cout << "[MOTOR ARBITRATOR] winner changed, new winner: " << win->id << " | emphasis: " << win->emphasis << std::endl;
            try { sim->sim().setStringSignal("active_behavior", winner); } catch (...) {}
        }
    if (sim != nullptr && leftMotor >= 0 && rightMotor >= 0) {
        try {
            sim->sim().setJointTargetVelocity(leftMotor,  vL / 0.0975); //raggio ruota
            sim->sim().setJointTargetVelocity(rightMotor, vR / 0.0975);
        } catch (...) { std::cout << "Exception Banner" << std::endl; }
    }
    if (win) {
        static std::string prevGoal;
        if (win->id != prevGoal) {
            prevGoal = win->id;
            std::map<std::string, double> best;
            for (const auto& p : proposals_)
                if (p.exclusive) best[p.id] = std::max(best[p.id], p.emphasis);
            std::vector<std::pair<std::string, double>> comp(best.begin(), best.end());
            DecisionLogger::get().log(SharedPerception::get().time(), win->id, win->emphasis, comp);
        }
    }
}

#include "ReachColor.h"
#include "SharedPerception.h"
#include <iostream>
#include <cmath>
#include <algorithm>

static double hueOf(const std::string& c) { 
    if (c == "Red") return 0.0;   
    if (c == "Orange") return 30.0; 
    if (c == "Yellow") return 60.0; 
    if (c == "Green") return 120.0; 
    return -1.0; 
}

static int countHue(const std::vector<unsigned char>& img, double hueDeg, double tol = 12.0) { 
    int n = 0; 
    for (size_t i = 0; i + 2 < img.size(); i += 3) { 
        const double r = img[i] / 255.0, g = img[i+1] / 255.0, b = img[i+2] / 255.0; 
        const double mx = std::max({r, g, b}), mn = std::min({r, g, b}), d = mx - mn; 
        if (mx < 0.25 || d / mx < 0.5) continue; 
        double h = (mx == r) ? 60.0 * std::fmod((g - b) / d, 6.0)
                 : (mx == g) ? 60.0 * ((b - r) / d + 2.0)
                             : 60.0 * ((r - g) / d + 4.0); 
        if (h < 0) h += 360.0; 
        const double dh = std::min(std::abs(h - hueDeg), 360.0 - std::abs(h - hueDeg)); 
        if (dh < tol) ++n; 
    } 
    return n; 
}

    if (target_ != previousTarget_) { 
        try { targetHandle_ = sim_->sim().getObjectHandle(target_); } catch(...) { targetHandle_ = -1; }
        previousTarget_ = target_; 
    } 
    return targetHandle_;

double ReachColor::readSensors() { 
    distance_ = 999.0; 
    isTargetVisible_ = false;

    getInput("Object", target_);
    if (target_.empty()) { tracking_ = false; return distance_; }

    targetHandle_ = resolveTarget();
    if (targetHandle_ == -1) return distance_;

    try {
        if (target_ == "/Bill") {
            auto bp = sim_->sim().getObjectPosition(targetHandle_, -1);
            const auto& rp = SharedPerception::get().position();
            const double yaw = SharedPerception::get().orientation()[2];
            const double dx = bp[0] - rp[0], dy = bp[1] - rp[1];
            relX_ =  std::cos(yaw) * dx + std::sin(yaw) * dy;
            relY_ = -std::sin(yaw) * dx + std::cos(yaw) * dy;
            distance_ = std::hypot(relX_, relY_);
            isTargetVisible_ = true;
            return distance_;
        }
        
        const int64_t cam = SharedPerception::get().cameraHandle();
        static double cachedT = -1.0;
        static std::vector<unsigned char> cachedImg;
        const double now = SharedPerception::get().time();
        if (now != cachedT) {
            auto res = sim_->sim().getVisionSensorImg(cam);
            const auto& img = std::get<0>(res);
            cachedImg.assign(img.begin(), img.end());
            cachedT = now;
        }

        std::string color; 
        getInput("Color", color);
        const double hue = hueOf(color);
        const double tol = (hue == 30.0 || hue == 60.0) ? 8.0 : 12.0;
        const int px = countHue(cachedImg, hue, tol);

        auto pos = sim_->sim().getObjectPosition(targetHandle_, robotHandle_);
        relX_ = pos[0]; 
        relY_ = pos[1];
        distance_ = std::hypot(relX_, relY_);

        isTargetVisible_ = (px >= 30) || (tracking_ && distance_ < 1.0);
        tracking_ = isTargetVisible_;
        
    } catch (const std::exception& e) {
        tracking_ = false;
        std::cout << "EXCEPTION IN REACH COLOR: " << e.what() << std::endl;
    }

    return distance_; 
}

bool ReachColor::checkInternalReleaser() { 
    return isTargetVisible_; 
}

std::pair<double, double> ReachColor::executeMotorSchema() {
    if (distance_ <= 0.55) return {0.0, 0.0};

    double angle = std::atan2(relY_, relX_);
    double v_max = 0.40;
    
    if (distance_ < 1.0)
        v_max = 0.15 + (distance_ / 1.0) * 0.25;

    double v = 0.0;
    if (std::abs(angle) < (M_PI / 3.0))
        v = v_max * std::cos(angle);

    double w = std::clamp(angle * 1.8, -1.2, 1.2);
    double L = 0.331;
    
    double v_left  = v - (w * L) / 2.0;
    double v_right = v + (w * L) / 2.0;

    return {v_left, v_right};
}
