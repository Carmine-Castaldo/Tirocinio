#ifndef ATTENTIONAL_BEHAVIOR_H
#define ATTENTIONAL_BEHAVIOR_H

#include <behaviortree_cpp/action_node.h>
#include <algorithm>
#include <string>
#include <utility>

#include "MotorCommand.h"
#include "SharedPerception.h"

class AttentionalBehavior : public BT::ThreadedAction {
protected:
    static constexpr double kDefaultMu   = 1.0;
    static constexpr double kMuStep      = 1.0;
    static constexpr double kMinLambda   = 0.01;
    static constexpr double kMinEmphasis = 1e-3;

    SharedPerception& sp;

    bool active_         = false;
    bool accomplished_   = false;
    double mu_           = kDefaultMu;
    double lambda_       = 1.0;
    double emphasis_     = 0.0;
    double nextActivation_ = 0.0;
    double lastVL_       = 0.0;
    double lastVR_       = 0.0;

    virtual void afterPropose() {}
    virtual bool accomplishesTask() const { return false; }
    virtual bool exclusive() const { return true; }


    static double saturatingLambda(double s, double rmin, double rmax,
                                   double lmin = kMinLambda, double lmax = 1.0) {
        if (s <= rmin) return lmin;
        if (s >= rmax) return lmax;
        return lmin + (lmax - lmin) * (s - rmin) / (rmax - rmin);
    }


    void resetState();
    virtual bool isFinished() const {
        return name() != "avoid" && lastVL_ == 0.0 && lastVR_ == 0.0;
    }
    void updateEmphasis();
    bool evaluateCycle(double now);
    void taskAccomplished();

    std::string getProposalID() const;
    void proposeCommand();
    void withdrawCommand();

public:
    AttentionalBehavior(const std::string& name, const BT::NodeConfig& config);
    ~AttentionalBehavior() override = default;

    static BT::PortsList providedPorts();

    virtual bool checkInternalReleaser() = 0;
    virtual bool checkExternalReleaser() = 0;
    virtual void updatePerceptualSchema() = 0;
    virtual std::pair<double, double> executeMotorSchema() = 0;

    std::string getUniqueID() const;

    BT::NodeStatus tick() override;
    void halt() override;
};

#endif // ATTENTIONAL_BEHAVIOR_H

class DecisionLogger {
public:
    static DecisionLogger& get() { static DecisionLogger l; return l; }

    void open(const std::string& path) {
        out_.open(path, std::ios::out | std::ios::trunc);
        if (out_.is_open()) out_ << "t,winner,e_winner,competitors\n";
    }

    void log(double t, const std::string& winner, double eWinner,
             const std::vector<std::pair<std::string, double>>& competitors) {
        if (!out_.is_open()) return;
        out_ << std::fixed << std::setprecision(2) << t << ',' << winner << ',' << eWinner << ",\"";
        for (size_t i = 0; i < competitors.size(); ++i) {
            if (i) out_ << ';';
            out_ << competitors[i].first << '=' << competitors[i].second;
        }
        out_ << "\"\n";
    }

    void close() { if (out_.is_open()) out_.close(); }

private:
    std::ofstream out_;
};

===== ./headers/SharedPerception.h =====
#pragma once

#include <chrono>
#include <condition_variable>
#include <cstdint>
#include <cmath>
#include <functional>
#include <map>
#include <algorithm>
#include <memory>
#include <mutex>
#include <string>
#include <tuple>
#include <unordered_map>
#include <unordered_set>
#include <utility>
#include <vector>

#include "RemoteAPIObjects.h"

class SharedPerception {
public:
    using Sonar = std::tuple<int64_t, double, std::vector<double>, int64_t, std::vector<double>>;

    static SharedPerception& get() {
        static SharedPerception instance;
        return instance;
    }
    SharedPerception(const SharedPerception&) = delete;
    SharedPerception& operator=(const SharedPerception&) = delete;

    void setup(RemoteAPIObjects* sim, int64_t robot, int64_t camera) {
        std::lock_guard<std::recursive_mutex> lock(m_);
        sim_    = sim;
        robot_  = robot;
        camera_ = camera;
    }

    template <class F>
    auto exec(F&& f) {
        std::lock_guard<std::recursive_mutex> lock(m_);
       //ProfScope prof(*this, "exec");
        return f(*sim_);
    }

    int64_t robotHandle() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        return robot_;
    }

    int64_t cameraHandle() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        return camera_;
    }


    void newTick(double simTime) {
        std::lock_guard<std::recursive_mutex> lock(m_);
        poseOk_ = false;
        treeOk_ = false;
        timeOk_ = false;

        worldPos_.clear();

        refreshPose();
        t_      = simTime;
        timeOk_ = true;

        updateBestGoal();

        ++seq_;
        done_    = 0;
        arbDone_ = false;
        cv_.notify_all();
    }

    void wake() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        cv_.notify_all();
        cvDone_.notify_all();
    }

    bool waitNextTick(uint64_t& seen, const std::function<bool()>& stop) {
        std::unique_lock<std::recursive_mutex> lock(m_);
        cv_.wait_for(lock, std::chrono::milliseconds(50), [&] {
            return seq_ > seen || stop();
        });
        if (seq_ > seen) {
            seen = seq_;
            return true;
        }
        return false;
    }

    void registerWorker() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        ++workers_;
    }

    void unregisterWorker() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        --workers_;
        cvDone_.notify_all();
    }

    void workerDone() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        ++done_;
        cvDone_.notify_all();
    }

    void waitWorkersDone(std::chrono::milliseconds timeout = std::chrono::milliseconds(50)) {
        std::unique_lock<std::recursive_mutex> lock(m_);
        cvDone_.wait_for(lock, timeout, [&] { return done_ >= workers_; });
    }

    void arbitratorDone() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        arbDone_ = true;
        cvDone_.notify_all();
    }

    void waitArbitratorDone(std::chrono::milliseconds timeout = std::chrono::milliseconds(50)) {
        std::unique_lock<std::recursive_mutex> lock(m_);
        cvDone_.wait_for(lock, timeout, [&] { return arbDone_; });
    }


    std::vector<double> position() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        refreshPose();
        return pos_;
    }

    std::vector<double> orientation() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        refreshPose();
        return ori_;
    }

    double time() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        if (!timeOk_) {
            t_      = sim_->sim().getSimulationTime();
            timeOk_ = true;
        }
        return t_;
    }

    Sonar sonar(int64_t h) {
        std::lock_guard<std::recursive_mutex> lock(m_);
        
        auto it = sonar_.find(h);
        if (it != sonar_.end()) {
            const Sonar& s    = it->second.reading;
            const bool near   = std::get<0>(s) == 1 && std::get<1>(s) < kSonarNear;
            const double maxAge = near ? 0.0 : kSonarIdleAge;
            if (t_ - it->second.time <= maxAge) return s;
        }
       //ProfScope prof(*this, "sonar");
        SonarEntry e{sim_->sim().readProximitySensor(h), t_};
        it = sonar_.insert_or_assign(h, std::move(e)).first;
        return it->second.reading;
    }

    bool inRobotTree(int64_t h) {
        std::lock_guard<std::recursive_mutex> lock(m_);
        if (!treeOk_) {
           //ProfScope prof(*this, "robotTree");
            tree_.clear();
            for (int64_t obj : sim_->sim().getObjectsInTree(robot_))
                tree_.insert(obj);
            
            treeOk_ = true;
        }
        return tree_.count(h) > 0;
    }

    std::shared_ptr<const std::vector<unsigned char>> image() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        if (!img_ || t_ - imgTime_ >= kImageMaxAge) {
           //ProfScope prof(*this, "image");
            sim_->sim().handleVisionSensor(camera_);
            auto r   = sim_->sim().getVisionSensorImg(camera_);
            img_     = std::make_shared<std::vector<unsigned char>>(std::move(std::get<0>(r)));
            imgTime_ = t_;
        }
        return img_;
    }

    int64_t handle(const std::string& path) {
        std::lock_guard<std::recursive_mutex> lock(m_);
        auto it = handles_.find(path);
        if (it != handles_.end())
            return it->second;

       //ProfScope prof(*this, "handle");
        const int64_t h = sim_->sim().getObjectHandle(path);
        handles_[path]  = h;
        return h;
    }

    std::vector<double> objectPosition(int64_t h, int64_t relativeTo) {
        std::lock_guard<std::recursive_mutex> lock(m_);
       //ProfScope prof(*this, "objPos");
        return sim_->sim().getObjectPosition(h, relativeTo);
    }

    int sonarSide(int64_t sensor) {
        std::lock_guard<std::recursive_mutex> lock(m_);
        auto it = sonarSide_.find(sensor);
        if (it == sonarSide_.end()) {
           //ProfScope prof(*this, "sonarSide");
            const auto p     = sim_->sim().getObjectPosition(sensor, robot_);
            const double ang = std::atan2(p[1], p[0]);
            it = sonarSide_.emplace(sensor, (ang > 0.35) ? +1 : (ang < -0.35 ? -1 : 0)).first;
        }
        return it->second;
    }

    std::vector<double> worldPosition(int64_t h) {
        std::lock_guard<std::recursive_mutex> lock(m_);
        auto it = worldPos_.find(h);
        if (it == worldPos_.end()) {
           //ProfScope prof(*this, "worldPos");
            it = worldPos_.emplace(h, sim_->sim().getObjectPosition(h, -1)).first;
        }
        return it->second;
    }

    double distanceTo(const std::string& path) {
        std::lock_guard<std::recursive_mutex> lock(m_);
       //ProfScope prof(*this, "distTo");
        return sim_->getDistanceTo(path);
    }

    void offerGoal(uint64_t key, int64_t handle, double emphasis) {
        std::lock_guard<std::recursive_mutex> lock(m_);
        if (handle >= 0)
            offers_[key] = {handle, emphasis};
    }

    void withdrawGoal(uint64_t key) {
        std::lock_guard<std::recursive_mutex> lock(m_);
        offers_.erase(key);
    }

    std::string profileReport() {
        std::lock_guard<std::recursive_mutex> lock(m_);
        std::vector<std::pair<std::string, ProfEntry>> v(prof_.begin(), prof_.end());
        prof_.clear();
        std::sort(v.begin(), v.end(), [](const auto& a, const auto& b) { return a.second.us > b.second.us; });
        std::string out;
        for (const auto& [tag, e] : v)
            out += tag + " n=" + std::to_string(e.n) + " " + std::to_string(e.us / 1000) + "ms | ";
        return out;
    }

    bool isGoal(int64_t detected) {
        std::lock_guard<std::recursive_mutex> lock(m_);
        if (goal_ < 0 || detected < 0) return false;
        if (detected == goal_) return true;

        if (goalTreeFor_ != goal_) {
           //ProfScope prof(*this, "goalTree");
            goalTree_.clear();
            try {
                for (int64_t obj : sim_->sim().getObjectsInTree(goal_)) {
                    goalTree_.insert(obj);
                }
            } catch (...) {}
            goalTreeFor_ = goal_;
        }
        return goalTree_.count(detected) > 0;
    }

private:
    static constexpr double kImageMaxAge = 0.2;
    
    static constexpr double kSonarNear    = 1.0;
    static constexpr double kSonarIdleAge = 0.3;

    struct ProfEntry { int n = 0; long long us = 0; };
    std::map<std::string, ProfEntry> prof_;

    struct ProfScope {
        SharedPerception& sp;
        const char* tag;
        std::chrono::steady_clock::time_point t0 = std::chrono::steady_clock::now();
        ProfScope(SharedPerception& s, const char* t) : sp(s), tag(t) {}
        ~ProfScope() {
            auto& e = sp.prof_[tag];
            ++e.n;
            e.us += std::chrono::duration_cast<std::chrono::microseconds>(
                        std::chrono::steady_clock::now() - t0).count();
        }
    };

    struct GoalOffer {
        int64_t handle  = -1;
        double emphasis = -1.0;
    };

    SharedPerception() = default;

    void refreshPose() {
        if (poseOk_) return;
        ProfScope prof(*this, "pose");
        
        const std::vector<double> q = sim_->sim().getObjectPose(robot_, -1);
        pos_    = {q[0], q[1], q[2]};
        const double yaw = std::atan2(2.0 * (q[6] * q[5] + q[3] * q[4]),
                                      1.0 - 2.0 * (q[4] * q[4] + q[5] * q[5]));
        ori_    = {0.0, 0.0, yaw};
        poseOk_ = true;
    }

    void updateBestGoal() {
        goal_ = -1;
        double bestEmphasis = -1.0;
        for (const auto& [uid, offer] : offers_) {
            if (offer.emphasis > bestEmphasis) {
                bestEmphasis = offer.emphasis;
                goal_        = offer.handle;
            }
        }
    }

    std::recursive_mutex m_;
    std::condition_variable_any cv_;
    std::condition_variable_any cvDone_;
    uint64_t seq_ = 0;
    int workers_  = 0;
    int done_     = 0;
    bool arbDone_ = false;

    RemoteAPIObjects* sim_ = nullptr;
    int64_t robot_         = -1;
    int64_t camera_        = -1;

    bool poseOk_ = false;
    bool treeOk_ = false;
    bool timeOk_ = false;
    std::vector<double> pos_;
    std::vector<double> ori_;
    double t_ = 0.0;
    struct SonarEntry { Sonar reading; double time = -1e9; };
    std::unordered_map<int64_t, SonarEntry> sonar_;
    std::unordered_map<int64_t, std::vector<double>> worldPos_;
    std::unordered_map<int64_t, int> sonarSide_;
    std::unordered_set<int64_t> tree_;
    std::shared_ptr<std::vector<unsigned char>> img_;
    double imgTime_ = -1e9;

    std::unordered_map<std::string, int64_t> handles_;
    std::unordered_map<uint64_t, GoalOffer> offers_;
    std::unordered_set<int64_t> goalTree_;
    int64_t goal_        = -1;
    int64_t goalTreeFor_ = -2;
};

#pragma once
#include <cmath>
#include <random>
#include <stdexcept>
#include <string>
#include <vector>
#include "RemoteAPIObjects.h"

inline unsigned& trialSeed() { static unsigned s = 1; return s; }

struct CubePose { std::string path; double x, y; };

inline std::vector<CubePose> randomizeCubes(RemoteAPIObjects& sim, unsigned seed,
                                            const std::vector<std::string>& cubePaths,
                                            const std::vector<std::string>& keepOutPaths,
                                            double half = 6.5,
                                            double minCubeDist = 1.5,
                                            double minKeepOut = 2.0)
{
    std::mt19937 rng(seed);
    std::uniform_real_distribution<double> U(-half, half);

    std::vector<std::pair<double, double>> keep;
    for (const auto& p : keepOutPaths) {
        auto pos = sim.sim().getObjectPosition(sim.sim().getObjectHandle(p), -1);
        keep.emplace_back(pos[0], pos[1]);
    }

    std::vector<CubePose> placed;
    for (const auto& path : cubePaths) {
        int64_t h = sim.sim().getObjectHandle(path);
        auto cur = sim.sim().getObjectPosition(h, -1);
        bool ok = false;
        for (int attempt = 0; attempt < 2000 && !ok; ++attempt) {
            const double x = U(rng), y = U(rng);
            ok = true;
            for (const auto& k : keep)   if (std::hypot(x - k.first, y - k.second) < minKeepOut) ok = false;
            for (const auto& c : placed) if (std::hypot(x - c.x, y - c.y) < minCubeDist)         ok = false;
            if (ok) {
                placed.push_back({path, x, y});
                sim.sim().setObjectPosition(h, -1, {x, y, cur[2]});
            }
        }
        if (!ok) throw std::runtime_error("randomizeCubes: nessuna posizione valida per " + path);
    }
    return placed;
}

int main(int argc, char** argv) {
    const unsigned seed   = (argc > 1) ? static_cast<unsigned>(std::stoul(argv[1])) : 1;
    const std::string out = (argc > 2) ? argv[2] : "results";
    const double maxT     = (argc > 3) ? std::stod(argv[3]) : 900.0;
    trialSeed() = seed;

    RemoteAPIClient client("127.0.0.1", 23000);
    RemoteAPIObjects sim(&client);

    SimHandles handles = initSimulation(sim);

    BT::BehaviorTreeFactory factory;
    auto tree = createBehaviorTree(factory, sim, "../plan/case1.xml");
    BT::TreeObserver observer(tree);
    auto blackboard = tree.rootBlackboard();

    setupEnvironmentAndBlackboard(sim, blackboard, seed, out);

    DecisionLogger::get().open(out + "/decisions_seed" + std::to_string(seed) + ".csv");
    sim.sim().startSimulation();

    auto [result, pathLen] = runSimulationLoop(client, sim, tree, handles, maxT);

    if (result != "SUCCESS") {
        logFailures(tree, observer, out, seed);
    }

    const double tEnd = SharedPerception::get().time();
    tree.haltTree();
    DecisionLogger::get().close();

    saveTrialSummary(out, seed, result, tEnd, pathLen);

    sim.sim().stopSimulation();

    return 0;
}



SimHandles initSimulation(RemoteAPIObjects& sim) {
    SimHandles h;
    h.leftMotor  = sim.sim().getObjectHandle((G_ROBOT_NAME + "/leftMotor").c_str());
    h.rightMotor = sim.sim().getObjectHandle((G_ROBOT_NAME + "/rightMotor").c_str());
    h.robot      = sim.sim().getObjectHandle(G_ROBOT_NAME.c_str());
    h.camera     = sim.sim().getObjectHandle((G_ROBOT_NAME + "/visionSensor").c_str());
    
    SharedPerception::get().setup(&sim, h.robot, h.camera);
    return h;
}

BT::Tree createBehaviorTree(BT::BehaviorTreeFactory& factory, RemoteAPIObjects& sim, const std::string& xmlPath) {
    factory.registerNodeType<AvoidObstacles>("AvoidObstacles", &sim);
    factory.registerNodeType<ReachColor>("ReachColor", &sim);
    factory.registerNodeType<GoToXY>("GoToXY", &sim);
    factory.registerNodeType<Place>("Place", &sim);
    factory.registerNodeType<Wander>("Wander", &sim);
    factory.registerNodeType<MotorArbitrator>("MotorArbitrator");
    
    return factory.createTreeFromFile(xmlPath);
}

void setupEnvironmentAndBlackboard(RemoteAPIObjects& sim, const BT::Blackboard::Ptr& blackboard, unsigned seed, const std::string& outDir) {
    auto cubes = randomizeCubes(sim, seed, G_CUBE_LIST, G_IGNORE_LIST);
    {
        std::ofstream f(outDir + "/cubes_seed" + std::to_string(seed) + ".csv");
        f << "cube,x,y\n";
        for (const auto& c : cubes) {
            f << c.path << ',' << c.x << ',' << c.y << '\n';
        }
    }

    for (const auto& c : cubes) {
        std::string cleanName = c.path.substr(1);         
        blackboard->set(cleanName + "_x", c.x);
        blackboard->set(cleanName + "_y", c.y);
    }

    int64_t billHandle = sim.sim().getObjectHandle(G_BILL_NAME.c_str());
    auto posBill = sim.sim().getObjectPosition(billHandle, -1);
    blackboard->set("Bill_x", posBill[0]);
    blackboard->set("Bill_y", posBill[1]);

    blackboard->set("Motor", MotorCommand{});
}

std::pair<std::string, double> runSimulationLoop(RemoteAPIClient& client, RemoteAPIObjects& sim, BT::Tree& tree, 
                                                 const SimHandles& handles, double maxT) {
    std::string result = "TIMEOUT";
    double pathLen = 0.0, px = 0.0, py = 0.0;
    bool havePrev = false;

    auto& sp = SharedPerception::get();
    double lastSimTime = -1.0;



    tree.tickExactlyOnce();
    std::this_thread::sleep_for(std::chrono::milliseconds(50));

    while (true) {
        const double simNow = sp.exec([](RemoteAPIObjects& s) { return s.sim().getSimulationTime(); });
        if (simNow <= lastSimTime) {
            std::this_thread::sleep_for(std::chrono::milliseconds(2));
            continue;
        }
        using clk = std::chrono::steady_clock;
        auto ms = [](clk::time_point a, clk::time_point b) {
            return std::chrono::duration_cast<std::chrono::milliseconds>(b - a).count();
        };

        const auto t0 = clk::now();
        const auto t1 = t0;

        lastSimTime = simNow;
        g_tickCount++;
        std::cout << "=============TICK " << g_tickCount << " START=============" << std::endl;
        sp.newTick(simNow);
        const auto t2 = clk::now();
        sp.waitArbitratorDone();
        const auto t3 = clk::now();
        
        BT::NodeStatus st = tree.tickExactlyOnce();
        const auto t4 = clk::now();
        std::cout << "=============TICK " << g_tickCount << " END=============" << std::endl;
        // std::cout << "[PROF] tick " << g_tickCount << " t=" << simNow
        //           << " | newTick=" << ms(t1, t2)
        //           << " waitArb=" << ms(t2, t3) << " treeTick=" << ms(t3, t4)
        //           << " TOTAL=" << ms(t0, t4) << " ms | " << sp.profileReport() << std::endl;

        const double t = sp.time();
        const auto pos = sp.position();
        if (havePrev)
            pathLen += std::hypot(pos[0] - px, pos[1] - py);

        px = pos[0];
        py = pos[1];
        havePrev = true;

        if (st != BT::NodeStatus::RUNNING) {
            result = (st == BT::NodeStatus::SUCCESS) ? "SUCCESS" : "FAILURE";
            break;
        }
        if (t > maxT) break;
    }

    return {result, pathLen};
}

void logFailures(const BT::Tree& tree, const BT::TreeObserver& observer, const std::string& outDir, unsigned seed) {
    std::vector<std::tuple<long long, std::string, unsigned>> rows;
    tree.applyVisitor([&](const BT::TreeNode* n) {
        const auto& ns = observer.getStatistics(n->UID());
        if (ns.last_result == BT::NodeStatus::FAILURE) {
            rows.emplace_back(ns.last_timestamp.count(), n->fullPath(), ns.failure_count);
        }
    });
    
    std::sort(rows.rbegin(), rows.rend());
    std::ofstream f(outDir + "/failures_seed" + std::to_string(seed) + ".txt");
    for (size_t i = 0; i < rows.size() && i < 15; ++i) {
        f << std::get<0>(rows[i]) << " ms  " << std::get<1>(rows[i])
          << "  failures=" << std::get<2>(rows[i]) << '\n';
    }
}

void saveTrialSummary(const std::string& outDir, unsigned seed, const std::string& result, double simTime, double pathLen) {
    const std::string summary = outDir + "/summary.csv";
    const bool exists = std::ifstream(summary).good();
    std::ofstream s(summary, std::ios::app);

    if (!exists) s << "seed,result,sim_time_s,path_m\n";
    s << seed << ',' << result << ',' << simTime << ',' << pathLen << '\n';
    
    std::cout << "[TRIAL " << seed << "] " << result 
              << " t=" << simTime << "s path=" << pathLen << "m\n";
}

#include "AttentionalBehavior.h"
#include <iostream>

AttentionalBehavior::AttentionalBehavior(const std::string& name, const BT::NodeConfig& config)
    : BT::ThreadedAction(name, config), sp(SharedPerception::get()) {}

BT::PortsList AttentionalBehavior::providedPorts() {
    return {
        BT::BidirectionalPort<double>("mu", kDefaultMu, "Top-Down Regulation mu_b"),
        BT::BidirectionalPort<MotorCommand>("Motor")
    };
}

void AttentionalBehavior::resetState() {
    nextActivation_ = 0.0;
    lastVL_         = 0.0;
    lastVR_         = 0.0;
    active_         = false;
}

void AttentionalBehavior::taskAccomplished() {
    if (accomplished_) return;
    accomplished_ = true;

    double currentMu = kDefaultMu;
    getInput("mu", currentMu);
    setOutput("mu", currentMu + kMuStep);
}

void AttentionalBehavior::updateEmphasis() {
    if (!getInput("mu", mu_)) mu_ = kDefaultMu;

    lambda_   = std::max(lambda_, kMinLambda);
    emphasis_ = mu_ / lambda_;
}

bool AttentionalBehavior::evaluateCycle(double now) {
    updatePerceptualSchema();
    updateEmphasis();

    active_ = checkInternalReleaser() && checkExternalReleaser();

    if (active_) {
        std::tie(lastVL_, lastVR_) = executeMotorSchema();

        if (isFinished()) {
            if (accomplishesTask()) {
                taskAccomplished();
            }
            return true;
        }
    } else {
        lastVL_ = 0.0;
        lastVR_ = 0.0;
    }

    nextActivation_ = now + 1.0 / std::max(emphasis_, kMinEmphasis);
    return false;
}

std::string AttentionalBehavior::getUniqueID() const {
    std::string obj, color, place;
    getInput("Object", obj);
    getInput("Color", color);
    getInput("Target", place);

    if (!exclusive()) 
        return name() + "_" + (obj == "/Bill" ? obj : "");
    
    std::string id = name();
    if (!obj.empty())   id += "_" + obj;
    if (!color.empty()) id += "_" + color;
    if (!place.empty()) id += "_" + place;
    return id;
}

std::string AttentionalBehavior::getProposalID() const {
    return getUniqueID() + "_" + std::to_string(UID());
}

void AttentionalBehavior::proposeCommand() {
    const std::string myID = getProposalID();
    
    std::lock_guard<std::mutex> lock(MotorCommand::lock());
    std::cout << "New Propose: " << myID << std::endl; 
    MotorCommand cmd;
    if (!getInput("Motor", cmd)) return;

    if (exclusive()) {
        if (cmd.owner == myID || emphasis_ > cmd.emphasis) {
            cmd.owner    = myID;
            cmd.emphasis = emphasis_;
            cmd.command  = {lastVL_, lastVR_};
            setOutput("Motor", cmd);
        }
    } else {
        if (cmd.avoidOwner == myID || emphasis_ >= cmd.avoidEmphasis) {
            cmd.avoidOwner    = myID;
            cmd.avoidEmphasis = emphasis_;
            cmd.avoidVL       = lastVL_;
            cmd.avoidVR       = lastVR_;
            setOutput("Motor", cmd);
        }
    }
}

void AttentionalBehavior::withdrawCommand() {
    const std::string myID = getProposalID();

    std::lock_guard<std::mutex> lock(MotorCommand::lock());
    MotorCommand cmd;
    if (!getInput("Motor", cmd)) return;

    if (exclusive()) {
        if (cmd.owner == myID) {
            cmd.owner    = "";
            cmd.emphasis = -1.0;
            cmd.command  = {0.0, 0.0};
            setOutput("Motor", cmd);
        }
    } else {
        if (cmd.avoidOwner == myID) {
            cmd.avoidOwner    = "";
            cmd.avoidEmphasis = -1.0;
            cmd.avoidVL       = 0.0;
            cmd.avoidVR       = 0.0;
            setOutput("Motor", cmd);
        }
    }
}

BT::NodeStatus AttentionalBehavior::tick() {
    const uint64_t key = UID();
    sp.registerWorker();

    struct WorkerScopeGuard {
        AttentionalBehavior* self;
        SharedPerception& sp;
        uint64_t key;
        ~WorkerScopeGuard() {
            self->withdrawCommand();
            sp.withdrawGoal(key);
            sp.unregisterWorker();
        }
    } guard{this, sp, key};

    uint64_t seen = 0;
    resetState();

    while (!isHaltRequested()) {
        if (!sp.waitNextTick(seen, [this] { return isHaltRequested(); }))
            continue;
        
        const double now = sp.time();

        try {
            if (now >= nextActivation_ && evaluateCycle(now)) {
                withdrawCommand();
                sp.workerDone();
                return BT::NodeStatus::SUCCESS;
            }

            if (active_) {
                proposeCommand();
                afterPropose();
            } else {
                withdrawCommand();
                sp.withdrawGoal(key);
            }
        } catch (const std::exception& e) {
            std::cout << "[" << name() << "] " << e.what() << '\n';
        }

        sp.workerDone();
    }

    return BT::NodeStatus::IDLE;
}

void AttentionalBehavior::halt() {
    sp.wake();
    BT::ThreadedAction::halt();
    withdrawCommand();
    sp.withdrawGoal(UID());
    resetState();
    accomplished_ = false;
}

AvoidObstacles::AvoidObstacles(const std::string& name, const BT::NodeConfig& config, RemoteAPIObjects* sim)
    : AttentionalBehavior(name, config), sim_(sim) {
    setupSensors();
}

BT::PortsList AvoidObstacles::providedPorts() {
    BT::PortsList ports = {
        BT::InputPort<std::string>("Object")
    };
    auto basePorts = AttentionalBehavior::providedPorts();
    ports.insert(basePorts.begin(), basePorts.end());
    return ports;
}

void AvoidObstacles::setupSensors() {
    auto& sp = SharedPerception::get();
    robotHandle_ = sp.handle("/PioneerP3DX");

    sonarHandles_.clear();
    sonarSide_.clear();
    for (int i = 1; i <= 6; ++i) {
        const int64_t h = sp.handle("/PioneerP3DX/ultrasonicSensor[" + std::to_string(i) + "]");
        sonarHandles_.push_back(h);
        sonarSide_.push_back(sp.sonarSide(h));
    }

    distances_.assign(sonarHandles_.size(), -1.0);
    lastDistances_.assign(sonarHandles_.size(), -1.0);
}

double AvoidObstacles::checkSonar(int64_t h) {
    const auto& result = SharedPerception::get().sonar(h);
    const int state = std::get<0>(result);
    
    if (state == 1) {
        const double dist = std::get<1>(result);
        const int64_t detectedObj = std::get<3>(result);
        
        if (!SharedPerception::get().isGoal(detectedObj) &&
            !SharedPerception::get().inRobotTree(detectedObj)) {
            return dist;
        }
    }
    return -1.0;
}

void AvoidObstacles::updateTargetHandle() {

void AvoidObstacles::updatePerceptualSchema() {
    updateTargetHandle();
    
    double minDistance = 999.0;
    isObstacle_ = false;

    for (size_t i = 0; i < sonarHandles_.size(); ++i) {
        distances_[i] = checkSonar(sonarHandles_[i]);
        if (distances_[i] > 0.0) {
            isObstacle_ = true;
            minDistance = std::min(minDistance, distances_[i]);
        }
    }

    const double now = SharedPerception::get().time();
    
    if (isObstacle_) {
        holdUntil_ = now + kHoldSec;
        lastDistances_ = distances_;
        lastMinDist_ = minDistance;
    } else if (now >= holdUntil_)
        turnSign_ = 0;
    
    double effectiveDist = isObstacle_ ? minDistance : lastMinDist_;
    lambda_ = saturatingLambda(effectiveDist, 0.5, 1.0);
}

std::pair<double, double> AvoidObstacles::executeMotorSchema() {
    const auto& d = isObstacle_ ? distances_ : lastDistances_;

    double turn = 0.0;
    double minDist = 999.0;
    double minCenterDist = 999.0;
    double leftSum = 0.0;
    double rightSum = 0.0;

    for (size_t i = 0; i < d.size(); ++i) {
        if (d[i] <= 0.0) continue;
        
        minDist = std::min(minDist, d[i]);
        double weight = 1.0 / (d[i] + kRobotRadius); 
        
        if (sonarSide_[i] > 0) { 
            turn -= weight; 
            leftSum += weight; 
        } else if (sonarSide_[i] < 0) { 
            turn += weight; 
            rightSum += weight; 
        } else
            minCenterDist = std::min(minCenterDist, d[i]); 
    }

    bool isSymmetric = std::abs(turn) < 0.3 * std::max(leftSum, rightSum);
    if (minCenterDist < 999.0 || isSymmetric) {
        if (turnSign_ == 0)
            turnSign_ = (leftSum > rightSum) ? -1 : +1;
        
        turn += turnSign_ * (1.0 / (std::min(minCenterDist, minDist) + kRobotRadius));
    }

    double linearVel = 0.175;
    double w_gain = 0.175;

    if (minDist < 0.20) {
        linearVel = 0.025;
        w_gain = 0.275;
    } else if (minDist < 0.40) 
        linearVel = 0.09;

    double w = std::clamp(turn * w_gain, -1.2, 1.2); 
    
    double v_left  = linearVel - (w * kWheelBase) / 2.0;
    double v_right = linearVel + (w * kWheelBase) / 2.0;

    return {v_left, v_right};
}

MotorArbitrator::ArbitratedOutput MotorArbitrator::resolveCommand() {
    ArbitratedOutput out;

    std::lock_guard<std::mutex> lock(MotorCommand::lock());
    MotorCommand cmd;
    if (!getInput("Motor", cmd)) {
        return out;
    }

    const bool hasWinner = !cmd.owner.empty() && cmd.emphasis > 0.0;
    const bool hasAvoid  = !cmd.avoidOwner.empty() && cmd.avoidEmphasis > 0.0;

    if (hasWinner && hasAvoid) {
        const double totalEmphasis = cmd.emphasis + cmd.avoidEmphasis;
        const double avoidRatio    = cmd.avoidEmphasis / totalEmphasis;
        const double goalRatio     = 1.0 - avoidRatio;

        out.vL       = (cmd.command.first  * goalRatio) + (cmd.avoidVL * avoidRatio);
        out.vR       = (cmd.command.second * goalRatio) + (cmd.avoidVR * avoidRatio);
        out.winner   = cmd.owner + "+avoid";
        out.emphasis = cmd.emphasis;
    } else if (hasWinner) {
        out.vL       = cmd.command.first;
        out.vR       = cmd.command.second;
        out.winner   = cmd.owner;
        out.emphasis = cmd.emphasis;
    } else if (hasAvoid) {
        out.vL       = cmd.avoidVL;
        out.vR       = cmd.avoidVR;
        out.winner   = "avoid";
        out.emphasis = cmd.avoidEmphasis;
    }

    return out;
}

void MotorArbitrator::applyToSimulator(const ArbitratedOutput& out,
                                       std::string& previousWinner,
                                       double& prevVL,
                                       double& prevVR) {
    const bool stopping   = out.vL == 0.0 && out.vR == 0.0 && (prevVL != 0.0 || prevVR != 0.0);
    const bool velChanged = stopping ||
                            std::abs(out.vL - prevVL) > kVelEpsilon ||
                            std::abs(out.vR - prevVR) > kVelEpsilon;
    const bool winChanged = out.winner != previousWinner;

    if (!velChanged && !winChanged) {
        return;
    }

    SharedPerception::get().exec([&](RemoteAPIObjects& s) {
        try {
            if (velChanged) {
                s.sim().setJointTargetVelocity(leftMotor_,  out.vL / kWheelRadius);
                s.sim().setJointTargetVelocity(rightMotor_, out.vR / kWheelRadius);
            }
            if (winChanged)
                s.sim().setStringSignal("active_behavior", out.winner);
            
        } catch (const std::exception& e) {
            std::cout << "[MotorArbitrator] error: " << e.what() << '\n';
        }
    });

    prevVL         = out.vL;
    prevVR         = out.vR;
    previousWinner = out.winner;
}

BT::NodeStatus MotorArbitrator::tick() {
    auto& sp = SharedPerception::get();
    std::string previousWinner = "none";
    double prevVL = 1e9;
    double prevVR = 1e9;
    uint64_t seen = 0;

    while (!isHaltRequested()) {
        if (!sp.waitNextTick(seen, [this] { return isHaltRequested(); }))
            continue;
        

        sp.waitWorkersDone(/*kWorkersTimeout*/);

        const ArbitratedOutput out = resolveCommand();
        applyToSimulator(out, previousWinner, prevVL, prevVR);

        sp.arbitratorDone();
    }

    return BT::NodeStatus::IDLE;
}

static double hueOf(const std::string& c) {
    if (c == "Red") return 0.0;   if (c == "Orange") return 30.0;
    if (c == "Yellow") return 60.0; if (c == "Green") return 120.0;
    return -1.0;
}

static int countHue(const std::vector<unsigned char>& img, double hueDeg, double tol = 12.0) {
    int n = 0;
    for (size_t i = 0; i + 2 < img.size(); i += 3) {
        const double r = img[i] / 255.0, g = img[i+1] / 255.0, b = img[i+2] / 255.0;
        const double mx = std::max({r, g, b}), mn = std::min({r, g, b}), d = mx - mn;
        if (mx < 0.25 || d / mx < 0.5) continue;
        double h = (mx == r) ? 60.0 * std::fmod((g - b) / d, 6.0)
                 : (mx == g) ? 60.0 * ((b - r) / d + 2.0)
                             : 60.0 * ((r - g) / d + 4.0);
        if (h < 0) h += 360.0;
        const double dh = std::min(std::abs(h - hueDeg), 360.0 - std::abs(h - hueDeg));
        if (dh < tol) ++n;
    }
    return n;
}

            lambda_ = saturatingLambda(distance_, 1.0, 10.0);
